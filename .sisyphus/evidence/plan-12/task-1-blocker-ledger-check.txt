Plan 12 Task 1 — Blocker Ledger Coverage Check
==============================================

Command
-------
python3 - <<'PY'
import json
from pathlib import Path
path = Path('.sisyphus/evidence/plan-12/blocker-ledger.json')
data = json.loads(path.read_text())
required_blockers = [
    'roadmap_overclaim_mismatch',
    'stale_authority_rationale',
    'circular_canonical_sources',
    'timestamp_precision_mismatch',
    'readme_mechanical_pass_overclaim',
    'f1_f3_rerun_reject',
    'historical_tmp_sqlite_incident',
    'tracked_audit_doc_scope_drift',
    'absolute_path_leakage',
]
required_fields = ['source_review', 'source_paths', 'disposition_plan', 'blocking']
blockers = {item['blocker_id']: item for item in data['blockers']}
missing_blockers = [bid for bid in required_blockers if bid not in blockers]
missing_required_fields = []
for bid in required_blockers:
    blocker = blockers.get(bid, {})
    missing = [field for field in required_fields if field not in blocker]
    if missing:
        missing_required_fields.append({'blocker_id': bid, 'fields': missing})
print(f'ledger_path={path}')
print(f'blocker_count={len(data["blockers"])}')
print(f'missing_blockers={missing_blockers}')
print(f'missing_required_fields={missing_required_fields}')
if missing_blockers or missing_required_fields:
    raise SystemExit(1)
PY
